class Controller(LeafSystem):
    """Defines a feedback controller for the double pendulum.

    The controller applies torques at the joints in order to:

    1) cancel out the dynamics of the double pendulum,
    2) make the first joint swing with the dynamics of a single pendulum, and
    3) drive the second joint towards zero.

    The magnitude of gravity for the imposed single pendulum dynamics is taken
    as a constructor argument.  So you can do fun things like pretending that
    gravity is zero, or even inverting gravity!
    """

    def __init__(self, multibody_plant, gravity):
        # 4 inputs (double pend state), 2 torque outputs.
        LeafSystem.__init__(self)
        self.DeclareVectorInputPort("double_pend_state", 4)
        self.DeclareVectorOutputPort("torque", 2, self.DoCalcVectorOutput)
        self.plant = multibody_plant
        self.g = gravity

    def DoCalcVectorOutput(self, context, torque):
        double_pend_state = self.get_input_port(0).Eval(context)
        # Extract manipulator dynamics.
        q = double_pend_state[:2]
        v = double_pend_state[-2:]
        (M, Cv, tauG, B, tauExt) = ManipulatorDynamics(self.plant, q, v)

        # Desired pendulum parameters.
        length = 2.0
        b = 0.1

        # Control gains for stabilizing the second joint.
        kp = 1
        kd = 0.1

        # Cancel double pend dynamics and inject single pend dynamics.
        torque.SetFromVector(
            Cv
            - tauG
            - tauExt
            + M.dot(
                [
                    self.g / length * np.sin(q[0]) - b * v[0],
                    -kp * q[1] - kd * v[1],
                ]
            )
        )

# 第三大部分：核心！反馈控制器代码（对应我们之前讲的u=B⁻¹(Cv−τG+M·目标加速度)）
## 3.1 自定义控制器类 `Controller(LeafSystem)`
### 类注释先看懂我们想要实现的3个效果（对应之前讲的目标）
```python
"""控制器要完成3件事：
1、抵消双摆原生的重力、杆子互相拉扯的复杂运动；
2、让肩膀关节像普通带阻尼单摆一样摆动；
3、把手肘关节稳定固定在竖直0角度，不让它乱晃
"""
```

### 构造函数 __init__：规定输入输出接口
```python
def __init__(self, multibody_plant, gravity):
    LeafSystem.__init__(self)
    # 输入接口：4个数字 [θ₁,θ₂,θ̇₁,θ̇₂] 双摆完整状态
    self.DeclareVectorInputPort("double_pend_state", 4)
    # 输出接口：2个数字 [τ₁,τ₂] 两台电机的扭力
    self.DeclareVectorOutputPort("torque", 2, self.DoCalcVectorOutput)
    self.plant = multibody_plant # 绑定双摆物理模型
    self.g = gravity # 传入重力系数
```
大白话：控制器程序有一个入口、一个出口；入口实时接收双摆当前角度速度，出口实时输出两台电机该拧多大劲。

### 核心计算函数 DoCalcVectorOutput（整条反馈抵消公式写在这里）
```python
def DoCalcVectorOutput(self, context, torque):
    # 1、读取从物理引擎传过来的双摆实时状态
    double_pend_state = self.get_input_port(0).Eval(context)
    q = double_pend_state[:2] # 拆分出两个角度 θ₁ θ₂
    v = double_pend_state[-2:] # 拆分出两个摆动速度 θ̇₁ θ̇₂

    # 2、调用工具算出当前这一刻的惯性、拉扯力、重力
    (M, Cv, tauG, B, tauExt) = ManipulatorDynamics(self.plant, q, v)

    # 自定义目标参数
    length = 2.0  # 虚拟单摆杆长
    b = 0.1       # 摆动阻尼，让杆子慢慢减速停下
    kp = 1        # 手肘位置修正系数，偏离0就拉回来
    kd = 0.1      # 手肘速度修正系数，晃太快就减速

    # 核心：计算两台电机需要输出的扭力
    torque.SetFromVector(
        Cv        # 抵消杆子互相拉扯的拖拽力
        - tauG    # 抵消重力往下拽的拉力
        - tauExt  # 抵消外部额外力（这里等于0，无作用）
        + M.dot(  # 乘以惯性矩阵，施加我们想要的目标加速度
            [
                # 肩膀目标加速度：单摆回落+阻尼减速
                self.g / length * np.sin(q[0]) - b * v[0],
                # 手肘目标加速度：拉回0角度、消除摆动速度，稳定不动
                -kp * q[1] - kd * v[1],
            ]
        )
    )
```
## 逐句对应我们之前推导的电机扭力公式，完美对上
之前推导公式（本双摆B是单位矩阵，B⁻¹=1，直接省略）：
$$\boldsymbol{u}= C\dot q - \tau_G + M \cdot \text{目标加速度}$$
代码里的计算顺序完全一模一样：
1. `Cv`：加一份反向力，抵消两根杆子互相拉扯；
2. `-tauG`：反向抵消重力下坠；
3. `+ M.dot(目标加速度向量)`：施加我们设定好的运动效果；
4. 最终算出的`torque`是2个数字：第一个τ₁肩膀电机出力，第二个τ₂手肘电机出力。

### 目标加速度两行分别解释（我们想要的效果）
1. 肩膀行 `self.g / length * np.sin(q[0]) - b * v[0]`
让肩膀杆像普通单摆，摆出去自动往回摆，同时带阻尼慢慢减速，不会一直来回晃。
2. 手肘行 `-kp * q[1] - kd * v[1]`
只要手肘偏离0竖直位置，就反向拉回；只要手肘有摆动速度，就减速；最终稳定在θ₂=0不动。
